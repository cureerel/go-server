# ============ USER TESTS ============

# Create user
curl -X POST http://localhost:8080/api/users \
  -H "Content-Type: application/json" \
  -d '{"name":"John Doe","email":"john@example.com","password":"secret123"}'

# Get all users
curl http://localhost:8080/api/users

# Get user by ID
curl http://localhost:8080/api/users/1

# Update user
curl -X PUT http://localhost:8080/api/users/1 \
  -H "Content-Type: application/json" \
  -d '{"name":"John Updated","email":"john.new@example.com"}'

# Delete user
curl -X DELETE http://localhost:8080/api/users/1

# ============ BLOG TESTS ============

# Create blog
curl -X POST http://localhost:8080/api/blogs \
  -H "Content-Type: application/json" \
  -d '{"title":"My First Post","content":"Hello World!","tags":"tech,go"}'

# Get all blogs
curl http://localhost:8080/api/blogs

# Get blog by ID
curl http://localhost:8080/api/blogs/1

# Get blog by slug (replace with actual slug from create response)
curl http://localhost:8080/api/blogs/slug/my-first-post-1234567890

# Update blog (PUT - full update)
curl -X PUT http://localhost:8080/api/blogs/1 \
  -H "Content-Type: application/json" \
  -d '{"title":"Updated Title","content":"Updated content","status":"published"}'

# Partial update (PATCH)
curl -X PATCH http://localhost:8080/api/blogs/1 \
  -H "Content-Type: application/json" \
  -d '{"status":"archived"}'

# Delete blog
curl -X DELETE http://localhost:8080/api/blogs/1
