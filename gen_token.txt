
package main

import (
	"fmt"
	"time"

	"github.com/golang-jwt/jwt/v4"
)

func main() {
	// Replace this with your server's actual secret (JWT_ACCESS_SECRET)
	secret := "your_server_secret_here"

	// User info
	claims := jwt.MapClaims{
		"user_id":    "21",
		"email":      "nikjurry2@gmail.com",
		"role":       "writer",
		"token_type": "access",
		"exp":        time.Now().Add(30 * time.Minute).Unix(), // expires in 30 minutes
		"iat":        time.Now().Unix(),
	}

	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)

	tokenString, err := token.SignedString([]byte(secret))
	if err != nil {
		panic(err)
	}

	fmt.Println(tokenString)
}
